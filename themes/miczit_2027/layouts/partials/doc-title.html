{{/* Text of the tag. Page titles are short ("Ollama"), so the full title is built here: the page, then its ancestors up to the top-level page (Home excluded), then the site title, e.g. "Ollama - Setup guides - ThunderAI - Mic's". The main page of an add-on (a top-level page with addon_status) also gets "Thunderbird Addon". Language redirect stubs have no Title: an ancestor stub borrows it from a translation, a page without Title shows the site title only. */}} {{ $parts := slice }} {{ if .Title }} {{ $chain := slice . }} {{ range .Ancestors }}{{ if not .IsHome }}{{ $chain = $chain | append . }}{{ end }}{{ end }} {{ range $chain }} {{ $src := . }} {{ if not .Title }}{{ range first 1 (where .Translations "Title" "!=" "") }}{{ $src = . }}{{ end }}{{ end }} {{ with $src.Title }}{{ $parts = $parts | append . }}{{ end }} {{ end }} {{ if and (eq (len $chain) 1) .Params.addon_status }}{{ $parts = $parts | append (i18n "thunderbirdAddon") }}{{ end }} {{ end }} {{ $parts = $parts | append site.Title }} {{ return delimit $parts " - " }}
